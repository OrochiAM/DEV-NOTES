CONTENT KOLEKCIJE - definicija izgleda podataka koje koristimo u astro fajlu

pise se u content.config.ts

import { defineCollection, z } from 'astro:content';
import { file, glob } from "astro/loaders";

const project = defineCollection({
  loader: file('src/data/projects.json'),
  loader: glob({ pattern: "**/*.md", base: "src/data/blog" }), - jos jedan primer ne ide dva
  schema: z.object({
    title: z.string(),
    description: z.string().optional(),
    description: z.string().max(200), 
    image: z.string(),
    slug: z.string(),
    liveSiteURL: z.string(),
    githubURL: z.string(),
    date: z.string().date(),
    tags: z.array(z.string()),
  })
});

export const collections = { project };

export const collections = { project, blog }; - vise kolekcija
